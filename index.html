-- High-Performance Recursive Common Table Expression (CTE)
-- Traverses hierarchy organizational parent-child tenant nodes,
-- calculating execution depth, run paths, and active budgets.
WITH RECURSIVE org_hierarchy AS (
    -- Anchor member
    SELECT 
        org_id, 
        parent_org_id, 
        org_name,
        1 AS depth, 
        org_name::text AS node_path,
        budget_allocated
    FROM organizations
    WHERE parent_org_id IS NULL AND is_active = true

    UNION ALL

    -- Recursive member joins back to anchor
    SELECT 
        child.org_id, 
        child.parent_org_id, 
        child.org_name,
        parent.depth + 1 AS depth,
        (parent.node_path || ' -> ' || child.org_name)::text,
        child.budget_allocated
    FROM organizations child
    INNER JOIN org_hierarchy parent ON child.parent_org_id = parent.org_id
    WHERE child.is_active = true
)
SELECT 
    org_id,
    parent_org_id,
    org_name,
    depth,
    node_path,
    budget_allocated,
    -- Apply window partition metrics
    SUM(budget_allocated) OVER (PARTITION BY parent_org_id) AS sibling_allocated,
    DENSE_RANK() OVER (PARTITION BY parent_org_id ORDER BY budget_allocated DESC) AS local_rank
FROM org_hierarchy
ORDER BY node_path;
-- Rigid Multi-Tenant Row-Level Security (RLS) Policy Configurations
-- Enforced directly at the bare PostgreSQL kernel boundary
ALTER TABLE client_transactions ENABLE ROW LEVEL SECURITY;

-- Block cross-tenant access explicitly for authenticated connections
CREATE POLICY tenant_isolation_assertion ON client_transactions
AS RESTRICTIVE
TO app_replica_instance, authenticated_role
USING (
    tenant_uuid = current_setting('app.current_tenant_id', true)
)
WITH CHECK (
    tenant_uuid = current_setting('app.current_tenant_id', true)
);

-- Compound indexing to accelerate isolated filters under high concurrency
CREATE INDEX CONCURRENTLY IF NOT EXISTS idx_transactions_tenant_composite
ON client_transactions (tenant_uuid, transaction_status, entity_hash_id);

-- Enforce transaction write safety lock check on isolated clusters
CREATE POLICY tenant_isolation_safe_insert ON client_transactions
FOR INSERT
TO app_replica_instance
WITH CHECK (
    tenant_uuid = current_setting('app.current_tenant_id', true)
    AND EXISTS (
        SELECT 1 FROM tenant_registry
        WHERE tenant_registry.id = tenant_uuid AND tenant_registry.is_active = true
    )
);
